def extract_text_from_pdf(pdf_file) -> str:
    """Extracts text from a PDF file using PyPDF2."""
    text = ""
    try:
        pdf_reader = PyPDF2.PdfReader(pdf_file)
        for page in pdf_reader.pages:
            page_text = page.extract_text()
            if page_text:
                text += page_text + "\n"
    except Exception as e:
        st.error(f"Error reading PDF: {e}")
    return text

def process_policy_document(text: str) -> List[dict]:
    """
    Extracts numbered clauses from the policy document.
    Returns a list of dictionaries with keys: clause_id, text, recommendations.
    """
    clauses = []
    clause_pattern = r'(\d+[\.\)]\s*)(.*?)(?=\d+[\.\)]|\Z)'
    matches = re.finditer(clause_pattern, text, re.DOTALL)
    for match in matches:
        clause_id = match.group(1).strip()
        clause_text = match.group(2).strip()
        if clause_text:
            clauses.append({"clause_id": clause_id, "text": clause_text, "recommendations": []})
    return clauses

def call_deepseek_api(prompt: str) -> str:
    """
    Calls the Azure DeepSeek API with the given prompt.
    Returns the AI response text.
    """
    try:
        headers = {
            "Content-Type": "application/json",
            "api-key": API_KEY
        }
        payload = {
            "messages": [
                {
                    "role": "system",
                    "content": "You are an AI assistant that categorizes policy recommendations based on policy clauses."
                },
                {
                    "role": "user",
                    "content": prompt
                }
            ],
            "temperature": 0.7,
            "top_p": 0.95,
            "max_tokens": 150
        }
        response = requests.post(AZURE_ENDPOINT, headers=headers, json=payload)
        if response.status_code == 200:
            result = response.json()
            content = result.get("choices", [{}])[0].get("message", {}).get("content", "")
            return content.strip()
        else:
            return f"API Error: {response.status_code}"
    except Exception as e:
        return f"Error: {str(e)}"

def generate_interactive_objective_chart(data: pd.Series, chart_type: str):
    """
    Generates an interactive Plotly chart for objective data.
    NA/blank values are included by filling them with the string "NA".
    """
    data = data.fillna("NA")
    counts = data.value_counts()
    df = pd.DataFrame({"Category": counts.index, "Count": counts.values})
    if chart_type == "bar":
        fig = px.bar(df, x="Category", y="Count", labels={"Category": "Category", "Count": "Count"})
    elif chart_type == "pie":
        fig = px.pie(df, names="Category", values="Count")
    elif chart_type == "line":
        fig = px.line(df, x="Category", y="Count", markers=True, labels={"Category": "Category", "Count": "Count"})
    elif chart_type == "doughnut":
        fig = px.pie(df, names="Category", values="Count", hole=0.4)
    else:
        fig = go.Figure()
    return fig

def generate_interactive_wordcloud(text: str):
    """
    Generates an interactive word cloud using Plotly.
    This function computes word frequencies and places words at random positions.
    """
    words = text.split()
    if not words:
        return None
    counter = Counter(words)
    df = pd.DataFrame(counter.items(), columns=["word", "frequency"])
    np.random.seed(42)
    df["x"] = np.random.rand(len(df))
    df["y"] = np.random.rand(len(df))
    fig = px.scatter(df, x="x", y="y", size="frequency", text="word",
                     hover_data={"frequency": True, "x": False, "y": False},
                     title="Interactive Word Cloud")
    fig.update_traces(textposition='top center')
    fig.update_layout(xaxis=dict(showgrid=False, zeroline=False, visible=False),
                      yaxis=dict(showgrid=False, zeroline=False, visible=False))
    return fig

def generate_interactive_circle_packing(data: pd.Series):
    """
    Generates an interactive bubble chart (as a placeholder for circle packing)
    using Plotly. NA/blank values are included.
    """
    data = data.fillna("NA")
    counts = data.value_counts()
    df = pd.DataFrame({"Category": counts.index, "Count": counts.values})
    np.random.seed(42)
    df["x"] = np.random.rand(len(df))
    df["y"] = np.random.rand(len(df))
    fig = px.scatter(df, x="x", y="y", size="Count", color="Category",
                     hover_name="Category", hover_data={"Count": True, "x": False, "y": False},
                     title="Interactive Circle Packing")
    fig.update_layout(xaxis=dict(showgrid=False, zeroline=False, visible=False),
                      yaxis=dict(showgrid=False, zeroline=False, visible=False))
    return fig

def map_clauses_to_feedback(clauses: List[dict], df: pd.DataFrame, subjective_columns: List[str]) -> List[dict]:
    """
    For each policy clause, check each subjective response (from the designated columns)
    and append the response to the clause's recommendations if keywords match.
    """
    for clause in clauses:
        recommendations = []
        for col in subjective_columns:
            # Do not filter out NA/blank values—treat them as "NA"
            for response in df[col].fillna("NA"):
                # Consider the first few words as keywords from the clause
                keywords = clause["text"].split()[:5]
                if any(keyword.lower() in response.lower() for keyword in keywords):
                    recommendations.append(response)
        clause["recommendations"] = recommendations
    return clauses
